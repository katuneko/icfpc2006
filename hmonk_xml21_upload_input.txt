run /bin/umodem xml21.adv EOF
{ O'Cult XML SNF normalizer, CPS traversal with global tag cleanup }

SNF x => Eval x Done;

{ small global cleanup rules }
Seq (Seq a b) c => Seq a (Seq b c);
Tag q (Tag q d) => Tag q d;
Tag Emph (Tag Bold d) => Tag Bold (Tag Emph d);
Tag Maj (Tag q d) => Tag q (Tag Maj d);

{ normalize children before rebuilding the document }
Eval (Seq x y) k => Eval x (KSeq y k);
Eval (Tag q d) k => Eval d (T q k);
Eval v k => k v;

(KSeq y k) v => Eval y (Merge v k);

Merge (Tag q a) k (Tag q b) => Eval (Seq a b) (T q k);
Merge a k b => k (Seq a b);

(T q k) v => k (Tag q v);

Done v => v;

. { end }
EOF
advise xml xml21.adv
