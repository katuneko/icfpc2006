{ O'Cult XML SNF normalizer (constructor continuations) }

SNF x => Eval x Done;

Seq (Seq a b) y => Seq a (Seq b y);

Eval (Seq x y) k => Eval x (KSeq y k);
Eval (t d) k => Eval d (Ins t k);
Eval v k => k v;

(KSeq y k) v => Eval y (Merge v k);

Merge (Tag q a) k (Tag q b) => Eval (Seq a b) (Ins (Tag q) k);
Merge a k b => k (Seq a b);

Ins t k (t d) => k (t d);
Ins (Tag Maj) k (Tag q d) => Ins (Tag Maj) (Ins (Tag q) k) d;
Ins e k (Tag Bold d) => Ins e (Ins (Tag Bold) k) d;
Ins t k d => k (t d);

Done v => v;

. { end }
EOF
