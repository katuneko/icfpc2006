{ XML SNF normalizer using Eval itself as the final continuation. }

SNF x => Eval Eval x;

Seq (Seq a b) y => Seq a (Seq b y);

Eval k (Seq x y) => Eval (KSeq y k) x;
Eval k (t d) => Eval (Ins t k) d;
Eval k => k;

(KSeq y k) v => Eval (Merge v k) y;

Merge (Tag q a) k (Tag q b) => Eval (Ins (Tag q) k) (Seq a b);
Merge a k b => k (Seq a b);

Ins t k (t d) => k (t d);
Ins (Tag Maj) k (Tag q d) => Ins (Tag Maj) (Ins (Tag q) k) d;
Ins e k (Tag Bold d) => Ins e (Ins (Tag Bold) k) d;
Ins t k d => k (t d);

.
EOF
